cmake_minimum_required(VERSION 3.20)
project(DriverRepairLoop LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This screen saver requires Windows.")
endif()
add_executable(DriverRepairLoop WIN32 main.cpp)
target_compile_features(DriverRepairLoop PRIVATE cxx_std_17)
target_compile_definitions(DriverRepairLoop PRIVATE UNICODE _UNICODE)
target_link_libraries(DriverRepairLoop PRIVATE user32 gdi32 shell32)
set_target_properties(DriverRepairLoop PROPERTIES SUFFIX ".scr")
if(MSVC)
  target_compile_options(DriverRepairLoop PRIVATE /utf-8 /W4)
endif()
